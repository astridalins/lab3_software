PRAGMA foreign_keys = ON;

CREATE TABLE users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(20) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    location VARCHAR(100),
    userType VARCHAR(100) NOT NULL,
    posicions VARCHAR(100),
    picture VARCHAR(255),
    colla VARCHAR(100)
);

CREATE TABLE admin (
    user_id INT PRIMARY KEY,
    is_admin INTEGER DEFAULT 1,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE post (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INT NOT NULL,
    parent_id INT,
    diada_id INT,
    private INTEGER DEFAULT 0,
    colla_name VARCHAR(100),
    text VARCHAR(500),
    image_path VARCHAR(500),
    created_at TEXT,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (parent_id) REFERENCES post(id) ON DELETE CASCADE
);

CREATE TABLE likes (
    user_id INT,
    post_id INT,
    PRIMARY KEY (user_id, post_id),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (post_id) REFERENCES post(id) ON DELETE CASCADE
);

CREATE TABLE follows (
    follower_id INT,
    followed_id INT,
    followed_at TEXT,
    PRIMARY KEY (follower_id, followed_id),
    FOREIGN KEY (follower_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (followed_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE TABLE colla (
    name VARCHAR(200) PRIMARY KEY
);

CREATE TABLE diada (

    created_at TEXT
);

CREATE TABLE review (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    diada_id INT NOT NULL,
    user_id INT NOT NULL,
    value INT,
    comment VARCHAR(255),
    espectador INTEGER NOT NULL,
    created_at TEXT,
    FOREIGN KEY (diada_id) REFERENCES diada(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
